program 	: M declar_list
	;

M 	:
	;

declar_list : declar_list declar
	 | declar
	 ;

declar	: var_declar
	| fun_declar
	;

var_declar : type_spec ID ";"
	;

type_spec : INT	{$$=INT;}
	| VOID	{$$=VOID;}
	| CHAR	{$$=CHAR;}
	;

fun_declar : type_spec fun_tag '(' LL params ')' comp_stmt
	| type_spec fun_tag '(' LL params ')' ';'
	;

LL : {level++;};

fun_tag	: callconvent ID
	| ID
	;

params	: param_list
	| VOID
	;

param_list : param_list ',' param
	| param
	;

param	: type_spec ID
	;

comp_stmt : '{' local_declars stmt_list '}'
	;

local_declars : local_declars var_declar
	|
	;

stmt_list	: stmt_list stmt
	|
	;

stmt	: expr_stmt
	| comp_stmt
	| if_stmt
	| while_stmt
	| return_stmt
	;

expr_stmt : expr ';'
	| ';'
	;

if_stmt	: IF '(' expr ')' stmt ELSE stmt
	| IF '(' expr ')' stmt
	;

while_stmt : WHILE '(' expr ')' stmt
	;

return_stmt : RETURN ';'
	| RETURN expr ';'
	;

expr	: var '=' expr
	| logic_expr
	;

var	: ID
	;

logic_expr : logic_expr OR logicand_expr
	| logicand_expr
	;

logicand_expr : logicand_expr AND simple_expr
	| simple_expr
	;

simple_expr : add_expr relop add_expr
	| add_expr
	;

add_expr	: add_expr ADDOP term
	| term
	;

term	: term mulop unary_expr
	| unary_expr
	;

unary_expr : unaryop unary_expr
	| factor
	;

factor	: '(' expr ')'
	| var
	| call
	| NUM
	| STRING
	;

call	: ID '(' args ')'
	;

args	: arg_list
	|
	;

arg_list	: arg_list ',' expr
	| expr
	;

callconvent : CDECL
	| STDCALL

relop	: LE
	| LT
	| GT
	| GE
	| EQ
	| NE
	;

mulop	: '*'
	| '/'
	| '%'
	;

unaryop	: '!'
	| '-'
	;
